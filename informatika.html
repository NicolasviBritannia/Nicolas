Panduan Integrasi Google Sheets (GAS)

Ikuti langkah berikut untuk menyambungkan aplikasi ini ke Google Spreadsheet Anda secara gratis:

  1. Buat Google Spreadsheet baru di Google Drive.
  2. Buka menu Extensions / Ekstensi > Apps Script.
  3. Hapus seluruh kode default, lalu salin dan tempel kode berikut:
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var data = [];
  
  for (var i = 1; i < rows.length; i++) {
    data.push({
      id: rows[i][0],
      tanggal: rows[i][1],
      jam: rows[i][2],
      kelas: rows[i][3],
      metode: rows[i][4],
      materi: rows[i][5],
      hadir: rows[i][6],
      sakit: rows[i][7],
      izin: rows[i][8],
      alfa: rows[i][9],
      refleksi: rows[i][10]
    });
  }
  
  var callback = e.parameter.callback;
  var output = JSON.stringify(data);
  if (callback) {
    return ContentService.createTextOutput(callback + '(' + output + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(output).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  var action = payload.action;
  var data = payload.data;
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Tanggal", "Jam", "Kelas", "Metode", "Materi", "Hadir", "Sakit", "Izin", "Alfa", "Refleksi"]);
  }
  
  if (action === "create") {
    sheet.appendRow([data.id, data.tanggal, data.jam, data.kelas, data.metode, data.materi, data.hadir, data.sakit, data.izin, data.alfa, data.refleksi]);
  } else if (action === "update") {
    var rows = sheet.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] == data.id) {
        sheet.getRange(i + 1, 1, 1, 11).setValues([[data.id, data.tanggal, data.jam, data.kelas, data.metode, data.materi, data.hadir, data.sakit, data.izin, data.alfa, data.refleksi]]);
        break;
      }
    }
  } else if (action === "delete") {
    var rows = sheet.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] == data.id) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
